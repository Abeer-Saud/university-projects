package edu.zuj.oop.week5.lecture2;

public class Circle {

    // public: could be accessed from anywhere
    // static: allocated once in app
    // final: value can't be modified
    // name should be ALL CAPS

    public static final double PI = 22/7;

    private int rad;

    // Constructor:
    // same as class name
    // no return datatype
    // can be called only using "new" keyword

    public Circle() {

    }

    // Constructor overloading:
    // having more than once constructor

    public Circle(int rad) {
        this.rad = rad;
    }

    public void getArea() {

        double area = PI * rad * rad;

        System.out.println("Area = " + area);
    }

    public void getRound() {

        double area = 2 * PI * rad;

        System.out.println("Round = " + area);
    }

    public int getRad() {
        return rad;
    }

    public void setRad(int rad) {

        // put local variable rad into the instance variable rad
        // this before variable allows us to access the INSTANCE variable

        if (rad > 0) {
            this.rad = rad;
        } else {
            System.out.println("Invalid rad " + rad);
        }
    }

    @Override
    public String toString() {

        return "Circle[" + "rad=" + rad + ']';
    }

}
